numbers = []

while True:
    print("\n--- Menu ---")
    print("1. Add")
    print("2. Remove")
    print("3. Display")
    print("4. Quit")

    choice = input("Enter your choice (1-4): ")

    # ADD
    if choice == "1":
        try:
            num = int(input("Enter an integer to add: "))
            numbers.append(num)
            print("Added successfully.")
        except:
            print("Invalid input")

    # REMOVE
    elif choice == "2":
        if not numbers:
            print("List is empty")
        else:
            num = int(input("Enter an integer to remove: "))
            if num in numbers:
                numbers.remove(num)
                print("Removed successfully.")
            else:
                print("Element not found")

    # DISPLAY
    elif choice == "3":
        if not numbers:
            print("List is empty")
        else:
            print("Current List:", numbers)

    # QUIT
    elif choice == "4":
        print("Exiting program...")
        break

    # INVALID OPTION
    else:
        print("Invalid choice")
